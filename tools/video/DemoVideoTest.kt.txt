/**
 * NOT part of the suite: copied in by tools/video/make_video.py, run once, and removed. One test per part of the app. Each starts from the same made-up ledger,
 * records the screen, and then really uses that part (typing, tapping, swiping) while its narration plays: every line waits as long as its clip lasts, and the
 * moment each began is written to the log (device clock) so the audio can be laid on the video afterwards.
 */
@RunWith(AndroidJUnit4::class)
class DemoVideoTest {
    @get:Rule val compose = createEmptyComposeRule()
    private val inst = InstrumentationRegistry.getInstrumentation()
    private val context = inst.targetContext
    private val container get() = (context.applicationContext as ExpensesApp).container
    private val ui = inst.uiAutomation
    private val dir = "/sdcard/Download"

    private fun sh(cmd: String): String = FileInputStream(ui.executeShellCommand(cmd).fileDescriptor).use { String(it.readBytes()) }
    private fun has(tag: String) = compose.onAllNodes(hasTestTag(tag), useUnmergedTree = true).fetchSemanticsNodes().isNotEmpty()
    private fun waitFor(tag: String) = compose.waitUntil(15_000) { has(tag) }
    private fun node(tag: String) = compose.onNodeWithTag(tag, useUnmergedTree = true)
    private fun tap(tag: String) { waitFor(tag); runCatching { node(tag).performScrollTo() }; node(tag).performClick(); pause(700) }
    private fun type(tag: String, text: String) { waitFor(tag); runCatching { node(tag).performScrollTo() }; runCatching { node(tag).performTextClearance() }; node(tag).performTextInput(text); pause(800) }
    private fun pause(ms: Long) = Thread.sleep(ms)
    private fun openMore(tile: String) { tap("more_menu"); pause(500); tap(tile) }

    private var durations = mapOf<String, Double>()
    private lateinit var scene: String

    private fun mark(key: String) { android.util.Log.w("DEMOMARK", "$scene $key ${System.currentTimeMillis()}") }

    /** Starts the line's narration, runs what the line is about while it plays, and waits out whatever is left of it. */
    private fun line(i: Int, actions: () -> Unit = {}) {
        val key = "${scene}_$i"
        mark(key); val t = System.currentTimeMillis()
        actions()
        val left = (durations.getValue(key) * 1000).toLong() + 600 - (System.currentTimeMillis() - t)
        if (left > 0) pause(left)
    }

    /** The same made-up ledger every time. The two texts (a bill and a bank debit) put banners above the list, so only the parts that are about them get them. */
    private fun seed(texts: Boolean) {
        runBlocking {
            container.startFresh()
            container.write { ok ->
                ok.store.saveAccount(Account(AccountId("hdfc"), "HDFC Card", AccountType.CREDIT_CARD, Currency.INR))
                val day = 86_400_000L; val now = System.currentTimeMillis()
                listOf(
                    Triple("Coffee", 180_00L, "cash" to "Food"), Triple("Auto fare", 120_00L, "cash" to "Transport"), Triple("Groceries run", 1_860_00L, "bank" to "Groceries"),
                    Triple("Dinner with friends", 1_240_00L, "hdfc" to "Food"), Triple("Metro recharge", 500_00L, "bank" to "Transport"), Triple("Phone recharge", 299_00L, "bank" to "Bills"),
                    Triple("Movie tickets", 700_00L, "hdfc" to "Fun"), Triple("Pharmacy", 415_00L, "bank" to "Health"), Triple("Coffee beans", 450_00L, "cash" to "Food"),
                    Triple("New shoes", 2_499_00L, "hdfc" to "Shopping"), Triple("Electricity", 1_842_00L, "bank" to "Bills"), Triple("Groceries top-up", 640_00L, "cash" to "Groceries"),
                ).forEachIndexed { i, (merchant, minor, where) ->
                    val (acct, cat) = where
                    ok.store.save(Expense(EntryId(UUID.randomUUID().toString()), now - (i * 1) * day - 3_600_000L * (i + 1), AccountId(acct), Money(minor, Currency.INR), listOf(CategorySplit(CategoryId(cat), Money(minor, Currency.INR)))), EntryDetails(merchant = merchant))
                }
            }
            if (!texts) return@runBlocking
            val due = LocalDate.now().plusDays(9).format(DateTimeFormatter.ofPattern("dd-MM-yyyy"))
            Capture.ingest(container, Capture.SMS, "AX-HDFCBK-S", "Your HDFC Bank Credit Card ending 4821 statement is generated. Total Amt Due Rs.8,450.00, Min Amt Due Rs.423.00, Due Date $due.", System.currentTimeMillis())
            Capture.ingest(container, Capture.SMS, "AX-HDFCBK-S", "Rs.1,250.00 debited from A/c XX1234 on ${LocalDate.now().format(DateTimeFormatter.ofPattern("dd-MM-yy"))} to VPA bigbasket@okhdfc (UPI Ref 628374659283). Not you? Call 18002586161", System.currentTimeMillis() + 1)
        }
    }

    /** One part of the app: the same ledger, a fresh recording, the actions, and the recording stopped. */
    private fun part(name: String, texts: Boolean = false, open: () -> Unit = {}, body: () -> Unit) {
        scene = name
        durations = sh("cat $dir/narr.txt").lines().filter { it.isNotBlank() }.associate { l -> l.substringBefore(' ') to l.substringAfter(' ').trim().toDouble() }
        sh("rm -f $dir/rec_$name.mp4")
        seed(texts)
        val s = ActivityScenario.launch<MainActivity>(Intent(context, MainActivity::class.java)); waitFor("add"); pause(1_200)
        mark("REC_START")
        ui.executeShellCommand("screenrecord --size 540x1170 --bit-rate 4000000 --time-limit 170 $dir/rec_$name.mp4")
        pause(1_500)
        try {
            // the narrator names the part while the app is taken to it
            val titleKey = "${name}_t"
            if (durations.containsKey(titleKey)) {
                mark(titleKey); val t = System.currentTimeMillis()
                open()
                val left = (durations.getValue(titleKey) * 1000).toLong() + 400 - (System.currentTimeMillis() - t)
                if (left > 0) pause(left)
            } else open()
            body()
        } finally {
            mark("END"); pause(800)
            sh("pkill -2 screenrecord"); pause(2_500) // finish the file properly
            s.close()
        }
    }
